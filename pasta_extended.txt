recipe Pasta_with_tomato_sauce {

    step 1 {
        get pot;
        add water 1 L to pot;
        add salt 10 g to pot;
    }

    step 2 {
        set num_of_people = 4;
        set sauce_type = "tomato";
    }

    step 3 {
        // This is a comment.

        if (num_of_people == 4) {
            get large_pan;
            add oil 20 g to large_pan;
        }

        if (sauce_type == "tomato") {
            add garlic 1 p to large_pan;
            add tomato 200 g to large_pan;
        }

        mix large_pan;
    }

    step 4 {
        serve Pasta_with_tomato_sauce;
    }
}
