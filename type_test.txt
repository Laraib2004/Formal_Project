// Type-checker demonstration.
// Every 'if' below compares values of incompatible types, so the compiler
// should report four type errors and finish with COMPILATION FAILED.

recipe Type_Mismatch_Test {
    step 1 {
        // Case 1: a holds a string, compared against an integer
        set a = "123";
        if (a == 123) {
            get pot;
        }
    }

    step 2 {
        // Case 2: b holds an integer, compared against a numeric string
        set b = 999;
        if (b == "999") {
            get pan;
        }
    }

    step 3 {
        // Case 3: string vs integer, both directions
        set name = "carbonara";
        set version = 1;

        if (name == 1) {
            get spoon;
        }

        if (version == "1") {
            get bowl;
        }
    }

    step 4 {
        serve Type_Mismatch_Test;
    }
}
