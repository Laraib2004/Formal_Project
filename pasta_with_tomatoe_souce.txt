recipe Pasta_with_tomato_sauce {

    step 1 {
        get pot;
        add water 1 L to pot;
        add salt 10 g to pot;
        cook water in pot 10 min;
        return pot;
    }

    step 2 {
        get pan;
        add oil 20 g to pan;
        add garlic 1 p to pan;
        cook garlic in pan 2 min;
        add tomato 200 g to pan;
        cook tomato in pan 15 min;
        return pan;
    }

    step 3 {
        get pot;
        add pasta 100 g to pot;
        boil pasta in pot 10 min;
        get colander;
        drain pasta colander;
        add pasta to pan;
        mix pan;
        serve Pasta_with_tomato_sauce;
    }
}
